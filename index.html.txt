  <script>
    const QUESTIONS = Array.from({ length: 20 }, (_, i) => {
      const multi = i % 3 === 1;
      return {
        id: i + 1,
        text: `Питання №${i + 1} (DELTA 16 AK): оберіть ${multi ? "ДВІ правильні відповіді" : "правильну відповідь"}.`,
        options: [
          `Варіант А для питання ${i + 1}`,
          `Варіант Б для питання ${i + 1}`,
          `Варіант В для питання ${i + 1}`,
          `Варіант Г для питання ${i + 1}`
        ],
        correct: multi ? [0, 2] : [1]
      };
    });

    const STORAGE_KEY = "delta_16_ak_results_v1";
    const ADMIN_PASS = "delta16";

    let currentStudent = null;
    let currentIndex = 0;
    let answers = Array.from({ length: QUESTIONS.length }, () => []);
    let isAdminAuth = false;

    function showView(view) {
      const isStudent = view === "student";
      document.getElementById("view-student").classList.toggle("hidden", !isStudent);
      document.getElementById("view-admin").classList.toggle("hidden", isStudent);

      document.getElementById("nav-student-btn").className = isStudent ? "btn-primary" : "btn-ghost";
      document.getElementById("nav-admin-btn").className = !isStudent ? "btn-primary" : "btn-ghost";

      if (!isStudent) {
        if (isAdminAuth) {
          renderAdmin();
        } else {
          document.getElementById("admin-pass").focus();
        }
      }
    }

    function startQuiz() {
      const nameInput = document.getElementById("student-name");
      const groupInput = document.getElementById("student-group");
      const errBox = document.getElementById("start-error");
      const name = nameInput.value.trim();
      const group = groupInput.value.trim();

      if (!name) {
        errBox.textContent = "Введіть прізвище та ініціали перед початком тесту.";
        nameInput.focus();
        return;
      }

      errBox.textContent = "";
      currentStudent = { name, group: group || "—" };
      currentIndex = 0;
      answers = Array.from({ length: QUESTIONS.length }, () => []);

      document.getElementById("start-card").classList.add("hidden");
      document.getElementById("result-card").classList.add("hidden");
      document.getElementById("quiz-card").classList.remove("hidden");
      renderQuestion();
    }

    function renderQuestion() {
      const q = QUESTIONS[currentIndex];
      const isMulti = q.correct.length > 1;
      const selected = answers[currentIndex];

      document.getElementById("quiz-error").textContent = "";
      document.getElementById("progress-fill").style.width = `${((currentIndex + 1) / QUESTIONS.length) * 100}%`;
      document.getElementById("q-num").textContent = `Питання ${currentIndex + 1} з ${QUESTIONS.length}`;
      document.getElementById("q-type").textContent = isMulti ? "Оберіть до 2 варіантів" : "1 правильна відповідь";
      document.getElementById("q-text").textContent = q.text;

      const box = document.getElementById("options-box");
      box.innerHTML = "";

      q.options.forEach((optText, idx) => {
        const isChecked = selected.includes(idx);
        const label = document.createElement("label");
        label.className = `opt${isChecked ? " selected" : ""}`;

        const input = document.createElement("input");
        input.type = isMulti ? "checkbox" : "radio";
        input.name = `q_${q.id}`;
        input.checked = isChecked;

        input.addEventListener("change", () => {
          handleOptionChange(idx, isMulti);
        });

        const span = document.createElement("span");
        span.textContent = optText;

        label.appendChild(input);
        label.appendChild(span);
        box.appendChild(label);
      });

      document.getElementById("prev-btn").disabled = currentIndex === 0;
      document.getElementById("next-btn").textContent =
        currentIndex === QUESTIONS.length - 1 ? "Завершити тест" : "Далі";
    }

    function handleOptionChange(idx, isMulti) {
      const errBox = document.getElementById("quiz-error");
      errBox.textContent = "";

      if (!isMulti) {
        answers[currentIndex] = [idx];
        renderQuestion();
        return;
      }

      const arr = answers[currentIndex];
      if (arr.includes(idx)) {
        answers[currentIndex] = arr.filter((x) => x !== idx);
        renderQuestion();
      } else {
        if (arr.length >= 2) {
          errBox.textContent = "У цьому питанні можна обрати максимум 2 варіанти.";
          renderQuestion();
          return;
        }
        answers[currentIndex] = [...arr, idx];
        renderQuestion();
      }
    }

    function prevQuestion() {
      if (currentIndex > 0) {
        currentIndex--;
        renderQuestion();
      }
    }

    function nextQuestion() {
      const errBox = document.getElementById("quiz-error");
      if (answers[currentIndex].length === 0) {
        errBox.textContent = "Оберіть хоча б один варіант відповіді.";
        return;
      }

      if (currentIndex < QUESTIONS.length - 1) {
        currentIndex++;
        renderQuestion();
      } else {
        finishQuiz();
      }
    }

    function isQuestionCorrect(q, userAns) {
      const a = [...q.correct].sort((x, y) => x - y).join(",");
      const b = [...userAns].sort((x, y) => x - y).join(",");
      return a === b;
    }

    function finishQuiz() {
      let score = 0;
      QUESTIONS.forEach((q, idx) => {
        if (isQuestionCorrect(q, answers[idx])) score++;
      });
      const pct = Math.round((score / QUESTIONS.length) * 100);

      const record = {
        name: currentStudent.name,
        group: currentStudent.group,
        score,
        total: QUESTIONS.length,
        pct,
        answers,
        date: new Date().toLocaleString("uk-UA")
      };

      const all = getResults();
      all.push(record);
      localStorage.setItem(STORAGE_KEY, JSON.stringify(all));

      document.getElementById("quiz-card").classList.add("hidden");
      document.getElementById("result-card").classList.remove("hidden");
      document.getElementById("res-name").textContent = `${record.name} (${record.group})`;
      document.getElementById("res-score").textContent = `${score} / ${QUESTIONS.length}`;
      document.getElementById("res-pct").textContent = `Точність: ${pct}%`;
    }

    function resetStudent() {
      document.getElementById("student-name").value = "";
      document.getElementById("student-group").value = "";
      document.getElementById("start-error").textContent = "";
      document.getElementById("result-card").classList.add("hidden");
      document.getElementById("quiz-card").classList.add("hidden");
      document.getElementById("start-card").classList.remove("hidden");
      document.getElementById("student-name").focus();
    }

    function getResults() {
      try {
        const raw = localStorage.getItem(STORAGE_KEY);
        const parsed = raw ? JSON.parse(raw) : [];
        return Array.isArray(parsed) ? parsed : [];
      } catch {
        return [];
      }
    }

    function loginAdmin() {
      const passInput = document.getElementById("admin-pass");
      const errBox = document.getElementById("admin-error");
      if (passInput.value === ADMIN_PASS) {
        isAdminAuth = true;
        errBox.textContent = "";
        passInput.value = "";
        document.getElementById("admin-login").classList.add("hidden");
        document.getElementById("admin-panel").classList.remove("hidden");
        renderAdmin();
      } else {
        errBox.textContent = "Невірний пароль. Спробуйте ще раз.";
        passInput.focus();
      }
    }

    function renderAdmin() {
      const results = getResults();
      const tbody = document.getElementById("results-tbody");
      tbody.innerHTML = "";

      const avgPct = results.length
        ? Math.round(results.reduce((s, r) => s + r.pct, 0) / results.length)
        : 0;

      document.getElementById("admin-summary").textContent =
        `Пройшло тестів: ${results.length} • Середній результат: ${avgPct}%`;

      if (!results.length) {
        const tr = document.createElement("tr");
        tr.innerHTML = `<td colspan="6" style="color:var(--muted); text-align:center;">Поки що немає проходжень тесту.</td>`;
        tbody.appendChild(tr);
      } else {
        results.slice().reverse().forEach((r, i) => {
          const tr = document.createElement("tr");
          tr.innerHTML = `
            <td>${results.length - i}</td>
            <td>${escapeHtml(r.name)}</td>
            <td>${escapeHtml(r.group)}</td>
            <td><strong>${r.score}/${r.total}</strong></td>
            <td>${r.pct}%</td>
            <td>${escapeHtml(r.date)}</td>
          `;
          tbody.appendChild(tr);
        });
      }

      renderAnalytics(results);
    }

    function renderAnalytics(results) {
      const box = document.getElementById("analytics-box");
      box.innerHTML = "";
      const totalAttempts = results.length;

      QUESTIONS.forEach((q, qIdx) => {
        let correctCount = 0;
        const optCounts = [0, 0, 0, 0];

        results.forEach((r) => {
          const userAns = (r.answers && r.answers[qIdx]) || [];
          if (isQuestionCorrect(q, userAns)) correctCount++;
          userAns.forEach((optIdx) => {
            if (optCounts[optIdx] !== undefined) optCounts[optIdx]++;
          });
        });

        const okPct = totalAttempts ? Math.round((correctCount / totalAttempts) * 100) : 0;
        const div = document.createElement("div");
        div.className = "analytics-q";
        div.innerHTML = `
          <div style="display:flex; justify-content:space-between; gap:8px;">
            <strong>${qIdx + 1}. ${escapeHtml(q.text)}</strong>
            <span>${correctCount}/${totalAttempts} (${okPct}%)</span>
          </div>
          <div class="bar-bg" style="margin:8px 0;"><div class="bar-ok" style="width:${okPct}%"></div></div>
          ${q.options.map((opt, oIdx) => {
            const isRight = q.correct.includes(oIdx);
            const cnt = optCounts[oIdx];
            const share = totalAttempts ? Math.round((cnt / totalAttempts) * 100) : 0;
            return `
              <div class="opt-stat">
                <span>${escapeHtml(opt)} ${isRight ? '<span class="correct-tag">[правильна]</span>' : ''}</span>
                <span>обрали: <strong>${cnt}</strong> (${share}%)</span>
              </div>
            `;
          }).join("")}
        `;
        box.appendChild(div);
      });
    }

    function exportCSV() {
      const results = getResults();
      if (!results.length) {
        alert("Немає даних для експорту.");
        return;
      }
      const rows = [["ПІБ", "Група", "Бали", "Всього", "Відсоток", "Дата"]];
      results.forEach((r) => rows.push([r.name, r.group, r.score, r.total, `${r.pct}%`, r.date]));
      const csv = rows
        .map((row) => row.map((cell) => `"${String(cell).replace(/"/g, '""')}"`).join(","))
        .join("\n");
      const blob = new Blob(["\uFEFF" + csv], { type: "text/csv;charset=utf-8;" });
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url;
      a.download = "delta_16_ak_results.csv";
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
    }

    function clearResults() {
      if (confirm("Видалити всі збережені результати студентів у цьому браузері?")) {
        localStorage.removeItem(STORAGE_KEY);
        renderAdmin();
      }
    }

    function escapeHtml(str) {
      return String(str).replace(/[&<>"']/g, (m) => ({
        "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;"
      }[m]));
    }

    document.getElementById("nav-student-btn").addEventListener("click", () => showView("student"));
    document.getElementById("nav-admin-btn").addEventListener("click", () => showView("admin"));
    document.getElementById("start-quiz-btn").addEventListener("click", startQuiz);
    document.getElementById("prev-btn").addEventListener("click", prevQuestion);
    document.getElementById("next-btn").addEventListener("click", nextQuestion);
    document.getElementById("restart-btn").addEventListener("click", resetStudent);
    document.getElementById("login-admin-btn").addEventListener("click", loginAdmin);
    document.getElementById("export-csv-btn").addEventListener("click", exportCSV);
    document.getElementById("clear-data-btn").addEventListener("click", clearResults);

    document.getElementById("student-name").addEventListener("keydown", (e) => {
      if (e.key === "Enter") startQuiz();
    });
    document.getElementById("student-group").addEventListener("keydown", (e) => {
      if (e.key === "Enter") startQuiz();
    });
    document.getElementById("admin-pass").addEventListener("keydown", (e) => {
      if (e.key === "Enter") loginAdmin();
    });
